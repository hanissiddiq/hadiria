CREATE OR REPLACE FUNCTION public.handle_permission_level1_approval()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  IF NEW.level = 1 AND NEW.status = 'Disetujui' THEN

    -- Buat approval level 2
    INSERT INTO public.permission_approvals (
      permission_request_id,
      level,
      status
    )
    VALUES (
      NEW.permission_request_id,
      2,
      'Menunggu'
    );

    -- Update status request
    UPDATE public.permission_requests
    SET
      status = 'Menunggu Level 2',
      updated_at = NOW()
    WHERE id = NEW.permission_request_id;

  END IF;

  RETURN NEW;
END;
$$;

create trigger trg_handle_permission_level1_approval
after
update on permission_approvals for EACH row
execute FUNCTION handle_permission_level1_approval ();

create view public.vlogbook
with
  (security_invoker = on) as
select
  a.id,
  a.user_id,
  a.attendance_date,
  a.shift,
  a.check_in,
  a.check_out,
  a.status,
  p.full_name,
  p."position",
  COALESCE(
    string_agg(
      t.task_name,
      '; '::text
      order by
        t.id
    ),
    '-'::text
  ) as uraian_kerja,
  l.description,
  l.activity_name
from
  attendances a
  join profiles p on p.id = a.user_id
  left join logbooks l on l.attendance_id = a.id
  left join tasks t on t.logbook_id = l.id
group by
  a.id,
  a.attendance_date,
  a.shift,
  a.status,
  p.full_name,
  p."position",
  l.description,
  l.activity_name;
