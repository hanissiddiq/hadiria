-- DROP FUNCTION public.batal_absen(int4);

CREATE OR REPLACE FUNCTION public.batal_absen(paramid integer)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
begin

  -- 1. Hapus TASKS (berdasarkan logbook yang terkait attendance)
  delete from tasks
  where logbook_id in (
    select id from logbooks where attendance_id = paramId
  );

  -- 2. Hapus LOGBOOK
  delete from logbooks
  where attendance_id = paramId;

  -- 3. Hapus ATTENDANCE
  delete from attendances
  where id = paramId;

end;
$function$
;



-- DROP FUNCTION public.generate_annual_quota(int4);

CREATE OR REPLACE FUNCTION public.generate_annual_quota(target_year integer DEFAULT NULL::integer)
 RETURNS text
 LANGUAGE plpgsql
AS $function$
DECLARE
    final_year INT;
    inserted_count INT;
BEGIN
    -- LOGIKA OTOMATIS: 
    -- Jika Frontend tidak kirim tahun (null), maka otomatis pakai tahun saat ini.
    -- Jika kirim angka (misal 2026), maka pakai angka tersebut.
    final_year := COALESCE(target_year, EXTRACT(YEAR FROM CURRENT_DATE)::INT);

    INSERT INTO master_leave_quota (user_id, year, annual_quota, used_leave)
    SELECT id, final_year, 12, 0
    FROM profiles
    WHERE NOT EXISTS (
        SELECT 1 FROM master_leave_quota 
        WHERE master_leave_quota.user_id = profiles.id 
        AND master_leave_quota.year = final_year
    )
    RETURNING count(*) INTO inserted_count;

    RETURN 'Sukses! ' || inserted_count || ' data kuota dibuat untuk tahun ' || final_year;
END;
$function$
;


-- DROP FUNCTION public.handle_level_2_approval(int8, uuid, text);

CREATE OR REPLACE FUNCTION public.handle_level_2_approval(p_leave_request_id bigint, p_approver_uuid uuid, p_status text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$DECLARE
  v_leave_request RECORD;
  v_leave_year INT;
  v_qr_code_url TEXT;
  v_domain TEXT := 'https://ppnpnberjaya.vercel.app'; 
  v_default_annual_quota NUMERIC := 12;
  v_already_counted BOOLEAN;
BEGIN
  -- 1. Ambil data cuti (termasuk kolom baru 'surat_sakit_url')
  SELECT * INTO v_leave_request
  FROM public.leave_requests
  WHERE id = p_leave_request_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Leave request not found.';
  END IF;

  v_already_counted := COALESCE(v_leave_request.annual_leave_counted, FALSE);

  -- 2. Proses jika 'Disetujui'
  IF p_status = 'Disetujui' THEN
    -- ... (Bagian 2a, 2b, 2c - Update status, QR, dan Approvals - tetap sama) ...
    UPDATE public.leave_requests
    SET 
      status = 'Disetujui', 
      approved_by = p_approver_uuid,
      annual_leave_counted = TRUE
    WHERE id = p_leave_request_id;

    v_qr_code_url := v_domain || '/leave/' || p_leave_request_id::TEXT;

    UPDATE public.leave_approvals
    SET 
      status = 'Disetujui', approved_at = NOW(), approver_id = p_approver_uuid,
      approved_by = p_approver_uuid, qr_code_url = v_qr_code_url where leave_request_id=p_leave_request_id and level=2;
      
    
    -- =================== INI LOGIKA YANG DIMODIFIKASI ===================
    -- 2d. Update kuota HANYA JIKA...
    IF (
        -- Skenario 1: Ini adalah 'Cuti Tahunan'
        v_leave_request.leave_type = 'Cuti Tahunan'
      ) OR (
        -- Skenario 2: Ini 'Cuti Sakit' TAPI TANPA surat dokter
        v_leave_request.leave_type = 'Cuti Sakit' AND v_leave_request.surat_sakit_url IS NULL
      )
    THEN
      -- Dan hanya jika belum pernah dihitung sebelumnya
      IF v_already_counted = FALSE THEN
        v_leave_year := EXTRACT(YEAR FROM v_leave_request.start_date);
        
        INSERT INTO public.master_leave_quota (user_id, year, annual_quota, used_leave)
        VALUES (
          v_leave_request.user_id, 
          v_leave_year, 
          v_default_annual_quota, 
          v_leave_request.durasi_hari_kerja
        )
        ON CONFLICT (user_id, year) DO UPDATE
        SET used_leave = master_leave_quota.used_leave + v_leave_request.durasi_hari_kerja;
      END IF;
      
    END IF;
    -- Jika leave_type = 'Cuti Sakit' DAN surat_sakit_url IS NOT NULL,
    -- blok IF di atas akan di-skip, dan kuota tidak akan terpotong.
    -- =================== AKHIR DARI MODIFIKASI ===================


  -- 3. Proses jika 'Ditolak' (Tidak berubah)
  ELSE
    UPDATE public.leave_requests
    SET status = 'Ditolak', annual_leave_counted = FALSE
    WHERE id = p_leave_request_id;

    -- INSERT INTO public.leave_approvals (
    --   leave_request_id, approver_id, level, status, approved_at, approved_by
    -- )
    -- VALUES (
    --   p_leave_request_id, p_approver_uuid, 2, 'Ditolak', NOW(), p_approver_uuid
    -- )
    -- ON CONFLICT (leave_request_id, level) DO 
    UPDATE public.leave_approvals
    SET 
      status = 'Ditolak', approved_at = NOW(), approver_id = p_approver_uuid,
      approved_by = p_approver_uuid, qr_code_url = NULL where leave_request_id=p_leave_request_id and level=2;
  END IF;

  RETURN jsonb_build_object('status', 'success', 'processed_status', p_status);

EXCEPTION
  WHEN OTHERS THEN
    RETURN jsonb_build_object('status', 'error', 'message', SQLERRM);
END;$function$
;


-- DROP FUNCTION public.handle_permission_approval(uuid, int8, text);

CREATE OR REPLACE FUNCTION public.handle_permission_approval(p_approver_uuid uuid, p_permission_request_id bigint, p_status text)
 RETURNS text
 LANGUAGE plpgsql
AS $function$
DECLARE
    current_level SMALLINT;
BEGIN
    -- 1️⃣ Cari level approval-nya
    SELECT level INTO current_level
    FROM public.permission_approvals
    WHERE approver_id = p_approver_uuid
      AND permission_request_id = p_permission_request_id
    LIMIT 1;

    IF current_level IS NULL THEN
        RAISE EXCEPTION 'Approver tidak ditemukan untuk izin ini.';
    END IF;

    -- 2️⃣ Update status approval sekarang
    UPDATE public.permission_approvals
    SET status = p_status,
        tanggal_persetujuan = NOW()
    WHERE approver_id = p_approver_uuid
      AND permission_request_id = p_permission_request_id;

    -- 3️⃣ Kalau level 1 disetujui → buat approval level 2
    IF current_level = 1 AND p_status = 'Disetujui' THEN
        INSERT INTO public.permission_approvals (permission_request_id, level, status)
        VALUES (p_permission_request_id, 2, 'Menunggu');
        RETURN 'Approval level 1 disetujui, menunggu level 2.';
    END IF;

    -- 4️⃣ Kalau level 2 disetujui → update status izin jadi Disetujui
    IF current_level = 2 AND p_status = 'Disetujui' THEN
        UPDATE public.permission_requests
        SET status = 'Disetujui'
        WHERE id = p_permission_request_id;
        RETURN 'Izin disetujui sepenuhnya.';
    END IF;

    -- 5️⃣ Kalau ada penolakan dari level mana pun
    IF p_status = 'Ditolak' THEN
        UPDATE public.permission_requests
        SET status = 'Ditolak'
        WHERE id = p_permission_request_id;
        RETURN 'Izin ditolak.';
    END IF;

    RETURN 'Status izin diperbarui.';
END;
$function$
;

-- DROP FUNCTION public.handle_permission_level1_approval();

CREATE OR REPLACE FUNCTION public.handle_permission_level1_approval()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
  IF (NEW.level = 1 AND NEW.status = 'Disetujui') THEN
    -- Buat approval level 2 otomatis
    INSERT INTO public.permission_approvals (permission_request_id, level, status)
    VALUES (NEW.permission_request_id, 2, 'Menunggu');

    -- Update status izin agar terlihat sedang menunggu level 2
    UPDATE public.permission_requests
    SET status = 'Menunggu Level 2',
        updated_at = NOW()
    WHERE id = NEW.permission_request_id;
  END IF;

  RETURN NEW;
END;
$function$
;

-- DROP FUNCTION public.submit_leave_request(uuid, text, date, date, text, text, bool, text, bool, text);

CREATE OR REPLACE FUNCTION public.submit_leave_request(p_user_id uuid, p_leave_type text, p_start_date date, p_end_date date, p_reason text, p_address text, p_half_day boolean, p_half_day_shift text, p_annual_leave_cut boolean, p_surat_sakit_url text)
 RETURNS TABLE(leave_request_id bigint, error_message text)
 LANGUAGE plpgsql
AS $function$DECLARE
    v_request_id bigint;
BEGIN
    -- 2. SEMUA LOGIKA VALIDASI DAN PERHITUNGAN HARI DIHAPUS DARI SINI
    --    Trigger 'simpan_snapshot_sisa_cuti' akan menanganinya.

    -- 3. Insert data, termasuk kolom baru dan HAPUS 'leave_days'
    INSERT INTO public.leave_requests(
        user_id, leave_type, start_date, end_date, reason, status,
        half_day, half_day_shift, address, annual_leave_cut, surat_sakit_url, leave_days, durasi_hari_kerja 
    ) VALUES (
        p_user_id, p_leave_type, p_start_date, p_end_date, p_reason, 'Menunggu',
        p_half_day, p_half_day_shift, p_address, p_annual_leave_cut, p_surat_sakit_url, 
        CASE 
            WHEN p_half_day = TRUE THEN 0.5
            ELSE (SELECT COUNT(*)
            FROM generate_series(
                p_start_date::date,
                p_end_date::date,
                '1 day'::interval
            ) AS d(day)
           -- WHERE EXTRACT(ISODOW FROM d.day) --NOT IN (6, 7)
            )
          END
          ,(SELECT COUNT(*)
            FROM generate_series(
                p_start_date::date,
                p_end_date::date,
                '1 day'::interval
            ) AS d(day)
           -- WHERE EXTRACT(ISODOW FROM d.day) --NOT IN (6, 7)
            )
    )
    RETURNING id INTO v_request_id;

    -- 4. Buat leave_approvals dengan ROLE yang benar (sesuai mentor Anda)
    --    Pastikan 'id' adalah kolom uuid di tabel 'profiles' Anda
    INSERT INTO public.leave_approvals(leave_request_id, approver_id, level, status)
    VALUES 
        (v_request_id, (SELECT id FROM public.profiles WHERE role='kasubbag' LIMIT 1), 1, 'Menunggu'),
        (v_request_id, (SELECT id FROM public.profiles WHERE role='kepala_kantor' LIMIT 1), 2, 'Menunggu');

    -- 5. Kembalikan ID request baru dan pesan sukses
    RETURN QUERY SELECT v_request_id, 'Pengajuan berhasil dikirim'::text;

EXCEPTION 
    -- Tangkap error APAPUN (termasuk dari trigger)
    WHEN OTHERS THEN
        RETURN QUERY SELECT NULL::bigint, SQLERRM::text;
END;$function$
;



-- DROP FUNCTION public.submit_permission_request(uuid, text, date, date, text, text, bool, text);

CREATE OR REPLACE FUNCTION public.submit_permission_request(p_user_id uuid, p_jenis_izin text, p_tanggal_mulai date, p_tanggal_selesai date, p_alasan text, p_lampiran_url text DEFAULT NULL::text, p_half_day boolean DEFAULT false, p_half_day_shift text DEFAULT NULL::text)
 RETURNS TABLE(request_id bigint, status text, error_message text)
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_new_id bigint;
    v_durasi int;
BEGIN
    -- Hitung durasi (Sederhana: selisih hari + 1)
    v_durasi := (p_tanggal_selesai - p_tanggal_mulai) + 1;

    -- Insert ke permission_requests
    INSERT INTO public.permission_requests (
        user_id, 
        jenis_izin, 
        tanggal_mulai, 
        tanggal_selesai, 
        alasan, 
        lampiran_url, 
        status, 
        durasi_hari_kerja,
        half_day,
        half_day_shift
    )
    VALUES (
        p_user_id, 
        p_jenis_izin, 
        p_tanggal_mulai, 
        p_tanggal_selesai, 
        p_alasan, 
        p_lampiran_url, 
        'Menunggu', -- FIX: Diubah dari 'Menunggu Persetujuan Kepala' menjadi 'Menunggu' agar sesuai aturan DB
        v_durasi,
        p_half_day,
        p_half_day_shift
    )
    RETURNING id INTO v_new_id;

    -- Return sukses
    RETURN QUERY SELECT v_new_id, 'success', 'Pengajuan izin berhasil dikirim';

EXCEPTION WHEN OTHERS THEN
    -- Return error jika ada masalah database
    RETURN QUERY SELECT NULL::bigint, 'error', SQLERRM;
END;
$function$
;
